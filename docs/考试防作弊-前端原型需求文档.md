#### ▲防作弊监控实现（前端模拟）

```javascript
// ===== 防作弊核心逻辑 =====

class ExamAntiCheat {
  constructor(config) {
    this.config = {
      maxSwitchScreen: 2,        // 最大切屏次数
      enableCopyPaste: false,    // 是否禁用复制粘贴
      enableRightClick: false,   // 是否禁用右键
      examDuration: 7200,        // 考试时长（秒）= 120分钟
      maxAttempts: 1,            // 限考次数
      ...config
    };

    this.state = {
      switchScreenCount: 0,      // 当前切屏次数
      remainingTime: this.config.examDuration,
      abnormalBehaviors: [],      // 异常行为记录
      isExamining: false,
      isForceSubmitted: false
    };
  }

  // ▲ 1. 切屏监控
  initScreenMonitor() {
    // 监听页面可见性变化
    document.addEventListener('visibilitychange', () => {
      if (document.hidden && this.state.isExamining) {
        this.state.switchScreenCount++;
        this.recordAbnormal('切屏操作', `第${this.state.switchScreenCount}次切屏`);

        // 更新UI显示
        this.updateStatusBar('切屏次数', `${this.state.switchScreenCount} / ${this.config.maxSwitchScreen}`);

        // 达到限制，强制交卷
        if (this.state.switchScreenCount >= this.config.maxSwitchScreen) {
          this.forceSubmit('切屏次数已达上限，系统已强制收卷');
        } else {
          // 显示警告弹窗
          this.showWarningModal(`警告：您已切屏 ${this.state.switchScreenCount} 次，剩余 ${this.config.maxSwitchScreen - this.state.switchScreenCount} 次机会，再次切屏将强制收卷！`);
        }
      }
    });
  }

  // ▲ 2. 复制粘贴禁用
  initCopyPasteBlock() {
    if (!this.config.enableCopyPaste) return;

    ['copy', 'cut', 'paste'].forEach(event => {
      document.addEventListener(event, (e) => {
        e.preventDefault();
        this.recordAbnormal('复制粘贴', `尝试${event}操作`);
        this.showToast('❌ 考试期间禁止复制粘贴操作！', 'error');
        return false;
      });
    });

    // 禁用Ctrl+C/V/X/A
    document.addEventListener('keydown', (e) => {
      if (e.ctrlKey || e.metaKey) {
        const blockedKeys = ['c', 'v', 'x', 'a'];
        if (blockedKeys.includes(e.key.toLowerCase())) {
          e.preventDefault();
          this.recordAbnormal('快捷键', `Ctrl+${e.key.toUpperCase()}`);
          this.showToast('❌ 该快捷键已被禁用！', 'error');
        }
      }
    });
  }

  // ▲ 3. 右键菜单禁用
  initRightClickBlock() {
    if (!this.config.enableRightClick) return;

    document.addEventListener('contextmenu', (e) => {
      e.preventDefault();
      this.recordAbnormal('右键菜单', '尝试打开右键菜单');
      this.showToast('❌ 考试期间禁止使用右键菜单！', 'error');
      return false;
    });
  }

  // ▲ 4. 全屏锁定（可选增强）
  initFullscreenLock() {
    const enterFullscreen = () => {
      const elem = document.documentElement;
      if (elem.requestFullscreen) elem.requestFullscreen();
      else if (elem.webkitRequestFullscreen) elem.webkitRequestFullscreen();
    };

    // 进入考试自动全屏
    enterFullscreen();

    // 监听全屏变化
    document.addEventListener('fullscreenchange', () => {
      if (!document.fullscreenElement && this.state.isExamining) {
        this.recordAbnormal('退出全屏', '尝试退出全屏模式');
        this.showWarningModal('警告：请勿退出全屏模式！');
        setTimeout(enterFullscreen, 1000);
      }
    });
  }

  // ▲ 5. 异常行为记录
  recordAbnormal(type, detail) {
    const record = {
      id: Date.now(),
      type,
      detail,
      timestamp: new Date().toLocaleString('zh-CN'),
      screenCount: this.state.switchScreenCount
    };
    this.state.abnormalBehaviors.push(record);
    this.updateStatusBar('异常行为', `${this.state.abnormalBehaviors.length} 条`);

    // 实时同步到监考端（模拟WebSocket）
    this.syncToMonitor(record);
  }

  // 强制交卷
  forceSubmit(reason) {
    this.state.isForceSubmitted = true;
    this.state.isExamining = false;
    this.showModal({
      title: '⚠️ 强制收卷',
      content: reason,
      okText: '确认并查看结果',
      onOk: () => {
        window.location.href = '/exam/result?forced=true';
      }
    });
  }

  // 倒计时
  startTimer() {
    this.state.isExamining = true;
    this.timer = setInterval(() => {
      this.state.remainingTime--;
      const formatted = this.formatTime(this.state.remainingTime);
      this.updateStatusBar('剩余时间', formatted);

      // 最后5分钟警告
      if (this.state.remainingTime === 300) {
        this.showWarningModal('⏰ 考试即将结束！剩余时间不足5分钟，请尽快完成答题。');
      }

      // 时间到自动交卷
      if (this.state.remainingTime <= 0) {
        clearInterval(this.timer);
        this.forceSubmit('考试时间已结束，系统自动交卷。');
      }
    }, 1000);
  }

  formatTime(seconds) {
    const h = Math.floor(seconds / 3600).toString().padStart(2, '0');
    const m = Math.floor((seconds % 3600) / 60).toString().padStart(2, '0');
    const s = (seconds % 60).toString().padStart(2, '0');
    return `${h}:${m}:${s}`;
  }
}
```
